name: Today Platform Regression

on:
  push:
    paths:
      - "index.html"
      - "sw.js"
      - "modules/**"
      - "tests/**"
      - "package.json"
      - "package-lock.json"
      - ".github/workflows/platform-regression.yml"
  pull_request:
    paths:
      - "index.html"
      - "sw.js"
      - "modules/**"
      - "tests/**"
      - "package.json"
      - "package-lock.json"
      - ".github/workflows/platform-regression.yml"

permissions:
  contents: read

concurrency:
  group: today-platform-regression-${{ github.ref }}
  cancel-in-progress: true

jobs:
  regression:
    name: Platform regression
    runs-on: ubuntu-latest
    timeout-minutes: 10

    steps:
      - name: Checkout
        uses: actions/checkout@v7

      - name: Use Node.js
        uses: actions/setup-node@v6
        with:
          node-version: "24"
          cache: npm

      - name: Install test dependencies
        run: npm ci --ignore-scripts

      - name: Run regression gate
        run: npm test
